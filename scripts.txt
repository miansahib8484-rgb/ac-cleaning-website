    // ── Wait for DOM
    document.addEventListener('DOMContentLoaded', function () {

      // ── Elements
      var hdr = document.getElementById('hdr');
      var floatWaEl = document.getElementById('floatWa');

      // ── Scroll handler
      var scrollTicking = false;
      window.addEventListener('scroll', function () {
        if (!scrollTicking) {
          window.requestAnimationFrame(function () {
            var sY = window.scrollY;
            updateActiveTabs(sY);
            if (hdr) hdr.classList.toggle('scrolled', sY > 20);
            if (floatWaEl) floatWaEl.classList.toggle('show', sY > 300);
            scrollTicking = false;
          });
          scrollTicking = true;
        }
      });

      // ── Show float-wa after delay
      setTimeout(function () { if (floatWaEl) floatWaEl.classList.add('show'); }, 1500);

      // ── Fade-in on scroll
      var io = new IntersectionObserver(function (entries) {
        entries.forEach(function (e) {
          if (e && e.isIntersecting && e.target) e.target.classList.add('in');
        });
      }, { threshold: 0.08 });
      document.querySelectorAll('.fade,.fade-l,.fade-r,.visible').forEach(function (el) {
        if (el) io.observe(el);
      });

      // ── Counter animation
      function animateCounter(el, target, suffix, duration) {
        var step = target / (duration / 16);
        var current = 0;
        var timer = setInterval(function () {
          current += step;
          if (current >= target) { current = target; clearInterval(timer); }
          el.innerHTML = Math.floor(current).toLocaleString() + '<sup>' + suffix + '</sup>';
        }, 16);
      }
      var counterObserver = new IntersectionObserver(function (entries) {
        entries.forEach(function (e) {
          if (e.isIntersecting && !e.target.dataset.counted) {
            e.target.dataset.counted = '1';
            var target = parseInt(e.target.dataset.target);
            var suf = e.target.dataset.suf || '';
            animateCounter(e.target, target, suf, 1800);
          }
        });
      }, { threshold: 0.3 });
      document.querySelectorAll('.stat-num[data-target]').forEach(function (el) {
        counterObserver.observe(el);
      });

      // ── Hamburger
      window.toggleMenu = function () {
        var menu = document.getElementById('mobMenu');
        var btn = document.getElementById('hambBtn');
        if (!menu || !btn) return;
        menu.classList.toggle('open');
        btn.classList.toggle('open');
        document.body.style.overflow = menu.classList.contains('open') ? 'hidden' : '';
      };

      // ── FAQ accordion
      window.toggleFaq = function (btn) {
        if (!btn) return;
        var item = btn.parentElement;
        if (!item) return;
        var isOpen = item.classList.contains('open');
        document.querySelectorAll('.faq-item.open').forEach(function (i) { if (i) i.classList.remove('open'); });
        if (!isOpen) item.classList.add('open');
      };
      window.toggleFaqHome = function (item) {
        if (!item) return;
        var isOpen = item.classList.contains('open');
        document.querySelectorAll('#faq .faq-item.open').forEach(function (i) { if (i) i.classList.remove('open'); });
        if (!isOpen) item.classList.add('open');
      };

      // ── Service tabs active on scroll
      var sections = ['ac-services', 'washing-machine', 'refrigerator', 'microwave', 'maintenance', 'emergency'];
      var tabs = document.querySelectorAll('.srv-tab');
      function updateActiveTabs(sY) {
        if (!tabs || !tabs.length) return;
        var current = '';
        var currentY = sY !== undefined ? sY : window.scrollY;
        sections.forEach(function (id) {
          var el = document.getElementById(id);
          if (el && currentY >= el.offsetTop - 160) current = id;
        });
        tabs.forEach(function (tab) {
          if (!tab) return;
          var isActive = tab.getAttribute('href') === '#' + current;
          if (isActive !== tab.classList.contains('active')) {
            if (isActive) tab.classList.add('active');
            else tab.classList.remove('active');
          }
        });
      }

      // ── Video controls
      window.playWhyVideo = function () {
        var v = document.getElementById('whyVideo');
        var overlay = document.getElementById('whyPlayOverlay');
        var poster = document.getElementById('whyPoster');
        var stopBtn = document.getElementById('whyStopBtn');
        if (v) v.play();
        if (overlay) overlay.classList.add('hidden');
        if (poster) poster.classList.add('hidden');
        if (stopBtn) stopBtn.classList.add('visible');
      };
      window.stopWhyVideo = function () {
        var v = document.getElementById('whyVideo');
        var overlay = document.getElementById('whyPlayOverlay');
        var poster = document.getElementById('whyPoster');
        var stopBtn = document.getElementById('whyStopBtn');
        if (v) { v.pause(); v.currentTime = 0; }
        if (overlay) overlay.classList.remove('hidden');
        if (poster) poster.classList.remove('hidden');
        if (stopBtn) stopBtn.classList.remove('visible');
      };

      // ── Smooth scroll
      document.querySelectorAll('a[href^="#"]').forEach(function (a) {
        a.addEventListener('click', function (e) {
          var href = this.getAttribute('href');
          if (href === '#') return;
          var target = document.querySelector(href);
          if (target) { e.preventDefault(); target.scrollIntoView({ behavior: 'smooth', block: 'start' }); }
        });
      });
